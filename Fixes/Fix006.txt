## 7. EDIT `DreamTasks.cs` — награда после толчка
Якорь (весь блок):
```
            else if (state == Protocol.Recover)
            {
                _wasRecover = true;
                if (bl[Bal.Support] > 0.5f)
                {

                    rew -= _penalties[3].Value;

            else
            {

                rew += T.PoseWeight * pose;
                if (_wasRecover) _rewards[3].Value = T.ReturnBonus;

```
Заменить на:
```
            else
            {
                // опора = промежуток между стопами независимо от контакта:
                // подвинуть маховую стопу к ξ выгодно сразу, долины нет
                float outside = XiOutside(out float spread);
                _maxSpread = MathF.Max(_maxSpread, spread);
                _penalties[3].Value = T.FootMarginCost * outside / 100f;
                rew -= _penalties[3].Value;

                if (state == Protocol.Recover) { _wasRecover = true; _recoverSec += dt; }
                else
                {
                    _rewards[2].Value = T.PoseWeight * pose;
                    _rewards[3].Value = T.ReturnBonus;          // за STAND после толчка, без условия _wasRecover
                    rew += T.PoseWeight * pose + T.ReturnBonus;
                }
            }

            _endPose = state == Protocol.Stand ? pose : 0f;
            if (tick >= _lastTick) Report(state == Protocol.Stand ? "СТОИТ" : "НЕ ВЕРНУЛАСЬ", tick);
            return rew;
        }

        float XiOutside(out float spread)
        {
            var g = G;
            float m = 0f, cx = 0f, cy = 0f, vx = 0f;
            foreach (var b in g.Bodies)
            {
                m += b.Mass;
                cx += b.Mass * b.GlobalPosition.X;
                cy += b.Mass * b.GlobalPosition.Y;
                vx += b.Mass * b.LinearVelocity.X;
            }
            cx /= m; cy /= m; vx /= m;
            var fa = g.Bodies[_footA].GlobalPosition;
            var fb = g.Bodies[_footB].GlobalPosition;
            float h = MathF.Max(MathF.Max(fa.Y, fb.Y) - cy, 20f);   // Y вниз, пол около 0
            float xi = cx + vx / MathF.Sqrt(980f / h);
            float lo = MathF.Min(fa.X, fb.X) - T.FootPad;
            float hi = MathF.Max(fa.X, fb.X) + T.FootPad;
            spread = MathF.Abs(fa.X - fb.X);
            return MathF.Max(0f, MathF.Max(lo - xi, xi - hi));
        }

        void Report(string how, int tick)
        {
            if (_reported || _examIndex < 0) return;
            _reported = true;
            DreamLog.Line($"  exam{_examIndex} толчок {_imp / T.PushMax:+0%;-0%} {how} t={tick / (float)Engine.PhysicsTicksPerSecond:0.00}s " +
                          $"recover={_recoverSec:0.00}s шаг={_maxSpread:0}px acute={_maxAcute:0.00} поза={_endPose:0.00}");
```
Внимание: у старого `Step` закрывающая скобка `}` идёт после `return rew;`. После замены она закрывает `Report`.
Проверить, что скобки сошлись. Если `g.Bodies` не содержит `RigidBody2D` (нет `Mass` / `LinearVelocity`) — СТОП.

## 8. CREATE `DreamLog.cs`
```csharp
using Godot;

public static class DreamLog
{
    const string Path = "user://dream_log.txt";
    public static void Line(string s)
    {
        GD.Print(s);
        using var f = FileAccess.FileExists(Path)
            ? FileAccess.Open(Path, FileAccess.ModeFlags.ReadWrite)
            : FileAccess.Open(Path, FileAccess.ModeFlags.Write);
        if (f == null) return;
        f.SeekEnd();
        f.StoreLine(s);
    }
}
```

Если у остальных классов в проекте есть namespace — положить DreamLog в тот же namespace.
