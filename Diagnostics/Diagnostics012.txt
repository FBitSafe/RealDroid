0. Разведка

Продолжая незавершённую работу Fix016, в `MotorRom.SetSwingTargets` уже были
внесены ограничения до начала этой сессии: достижимая длина ноги 0.95,
`SwingKneeMax`, запас hip 0.05 рад и посадочная фаза. Полный код функции,
проверенный в ходе этой работы:

    void SetSwingTargets(Ragdoll body, Brain brain, int foot, float targetX, float targetY, float progress)
    {
        bool near = foot == body.FootNear;
        string suffix = near ? "n" : "f";
        int hip = body.Def.JointIndex($"hip_{suffix}");
        int knee = body.Def.JointIndex($"knee_{suffix}");
        int ankle = body.Def.JointIndex($"ankle_{suffix}");
        if (hip < 0 || knee < 0 || ankle < 0) return;

        Vector2 hipPoint = body.JointPos(hip);
        var ankleDef = body.Def.Joints[ankle];
        var footDef = body.Def.Parts[foot];
        Vector2 ankleOffset = ankleDef.Anchor - footDef.Center;
        Vector2 ankleTarget = new(targetX + ankleOffset.X, targetY + ankleOffset.Y);
        Vector2 toTarget = ankleTarget - hipPoint;
        float thighLength = body.Def.Parts[body.Def.PartIndex(near ? "thigh_n" : "thigh_f")].Length;
        float shinLength = body.Def.Parts[body.Def.PartIndex(near ? "shin_n" : "shin_f")].Length;
        float maxReach = 0.95f * (thighLength + shinLength);
        float distance = Math.Clamp(toTarget.Length(), 1f, maxReach);
        if (toTarget.Length() > maxReach)
            toTarget = toTarget.Normalized() * maxReach;
        float kneeAngle = MathF.Acos(Math.Clamp(
            (thighLength * thighLength + shinLength * shinLength - distance * distance) /
            (2f * thighLength * shinLength), -1f, 1f));
        kneeAngle = Math.Clamp(kneeAngle, 0f, SwingKneeMax);
        float targetBearing = -MathF.Atan2(toTarget.X, toTarget.Y);
        float thighRotation = targetBearing - MathF.Atan2(
            shinLength * MathF.Sin(kneeAngle),
            thighLength + shinLength * MathF.Cos(kneeAngle));
        float pelvisRotation = body.Bodies[body.Pelvis].GlobalRotation;
        float thighTarget = Mathf.Wrap(thighRotation - pelvisRotation, -Mathf.Pi, Mathf.Pi);
        var hipDef = body.Def.Joints[hip];
        float hipLo = Math.Min(hipDef.Lower + 0.05f, hipDef.Upper - 0.05f);
        float hipHi = Math.Max(hipDef.Lower + 0.05f, hipDef.Upper - 0.05f);
        thighTarget = Math.Clamp(thighTarget, hipLo, hipHi);
        float kneeTarget = kneeAngle;
        float shinRotation = thighRotation + kneeAngle;
        float ankleTargetAngle = Mathf.Wrap(-shinRotation, -Mathf.Pi, Mathf.Pi);

        brain.MotorTarget[hip] = Math.Clamp(thighTarget, body.Def.Joints[hip].Lower, body.Def.Joints[hip].Upper);
        brain.MotorTarget[knee] = Math.Clamp(kneeTarget, body.Def.Joints[knee].Lower, body.Def.Joints[knee].Upper);
        brain.MotorTarget[ankle] = Math.Clamp(ankleTargetAngle, body.Def.Joints[ankle].Lower, body.Def.Joints[ankle].Upper);
        brain.MotorStiff[hip] *= SwingStiffness;
        brain.MotorStiff[knee] *= SwingStiffness;
        brain.MotorStiff[ankle] *= SwingStiffness;
        if (progress >= 1f - LandPhase)
        {
            brain.MotorStiff[hip] = LandStiff;
            brain.MotorStiff[knee] = LandStiff;
            brain.MotorTarget[knee] = Math.Clamp(
                Math.Max(brain.MotorTarget[knee], LandKnee),
                body.Def.Joints[knee].Lower, body.Def.Joints[knee].Upper);
        }
        if (progress < 1f)
        {
            brain.SwingingJoints[hip] = true;
            brain.SwingingJoints[knee] = true;
            brain.SwingingJoints[ankle] = true;
        }
    }

Исходная посадочная фаза Fix015 сохраняла цели бедра/колена и удерживала
жёсткость на LandStiff, пока стопа не касалась пола. При контакте жёсткость
переходила к базовой за LandRamp; завершение дуги без контакта оставляло ногу
в посадочном состоянии навсегда. В настоящем исправлении посадочная цель Y
равна `_swingStartY + LandPress`.

1. Выполненные изменения

- Посадка без контакта ограничена LandTimeout=0.2 с: целевое колено ведётся к
  LandKnee, бедро и голеностоп — к базовой позе, жёсткость всех трёх суставов
  — к базовой. Таймаут завершает шаг, снимает SwingingJoints и учитывается в
  CapturePointTimeouts.
- Контакт завершает шаг после LandRamp. CapturePointTimeouts учитываются
  отдельно от CapturePointSteps.
- Цель стопы ограничивается расстоянием 0.95 от суммы длин бедра и голени;
  цель hip ограничена суставными пределами с inset 0.05 рад; цель колена при
  переносе ограничена SwingKneeMax=1.2 рад. Геометрический расчёт предельного
  горизонтального выноса выполнен в Reset один раз на ROM-чип.
- Если шаг закончился с ξ за границей опоры, другая нога может шагнуть после
  RearmDelay=0.08 с; серия ограничена тремя шагами и сбрасывается при возврате
  ξ в опору.
- После 0.3 с внутри опоры при двух стопах на полу и spread больше ExitSpread
  выполняется один приставной шаг дальней от ξ ногой к другой стопе с зазором
  5 px. Повторное срабатывание блокируется до нового выхода ξ за опору.
- Добавлен --levels с четырьмя экзаменами на уровень. Экзамен проходит, если
  персонаж не упал и RecoverTask вернул его в STAND. Тест запускает MOTOR ROM
  без загрузки/сохранения нейронных весов. Swing-leg reflex protection включена
  по умолчанию.
- Вычисленный геометрический горизонтальный вынос: BackStepMax=62.0 px,
  FwdStepMax=82.0 px.

3. Постепенное уменьшение CP-step assist в NN training

По запросу добавлена линейная интерполяция между базовым MOTOR NN/ROM выходом
и MOTOR ROM capture-point step. `--cp-step-fade-gens N` управляет длиной
затухания; по умолчанию N=100 Recover-поколений. На поколении 0 помощь равна
100%, затем линейно уменьшается до 0% к указанному номеру поколения; значение
0 отключает помощь сразу. Счётчик Recover-поколений берётся из
`SectorGen[RECOVER]`, включая локальный прогресс текущего Recover-training
сеанса. Скаляр затухания применяется к целям и жёсткости суставов шага; NN
остаточные поправки остаются без изменений. Флаг `--no-cp-step` по-прежнему
немедленно отключает CP-step вне зависимости от расписания.

Флаг можно настроить в Godot Inspector через `CpStepFadeGenerations` или при
запуске, например:

    --sector RECOVER --cp-step-fade-gens 50

Начальный лог показывает текущую долю и длину расписания; лог каждого
поколения содержит `CP step NN%`.
